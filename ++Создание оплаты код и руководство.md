# Google Play Billing — Руководство для новых Capacitor-проектов
> Написано на основе боевого кода Neyra/Moodos. Давай в новую сессию без изменений.

---

## Стек

- Capacitor 8+
- `capacitor-plugin-cdv-purchase` v13+ (cordova-plugin-purchase с Capacitor-обёрткой)
- Google Play Billing Library 7.x (внутри плагина)
- Vanilla JS / ES Modules (без фреймворков)

---

## Архитектура: три слоя

```
Google Play (источник истины)
        ↓
billing-service.js   ← единственный файл, работающий со store
        ↓
user-profile.js      ← хранит флаги, isPremium(), активация/деактивация
        ↓
state-execution-engine.js  ← применяет состояние к UI (один единственный вход)
```

**Правило:** никакой другой файл не должен обращаться к `store` напрямую. Только `billing-service.js`.

## Шаг 2. Google Play Console

1. Приложение → Монетизация → Подписки
2. Создать два продукта:
   - ID: `premium_monthly`
   - ID: `premium_yearly`
3. Установить цены, активировать
4. **Важно:** опубликовать хотя бы в Internal Testing — без этого продукты не загружаются
5. Настройки → Тестирование → Лицензионное тестирование → добавить Gmail тестировщиков

**Тестовые подписки:** monthly = 5 минут/цикл, yearly = 30 минут/цикл. Автоотмена после 6 продлений.

## Шаг 3. billing-service.js

```js
import { activatePremiumPaid, deactivateExpiredPremium, restorePremiumFromProfile, getProfile } from './user-profile.js';
import { enqueueBillingStateUpdate } from '../core/event-queue.js';
import { store, ProductType, Platform } from 'capacitor-plugin-cdv-purchase';

let storeReady = false;

export function isStoreReady() {
  return storeReady;
}

export function initBilling() {
  // Защита от двойного вызова
  if (window._billingInitialized) return;
  window._billingInitialized = true;
  window._billingInitializing = true;

  // Временно восстановить из profile пока store грузится
  try {
    const profile = getProfile();
    if (profile?.premium_type === 'paid' && profile?.premiumExpiresAt > Date.now()) {
      window._trustedSetBillingPremium?.(true);
      if (window.systemState) window.systemState.premium = true;
    }
  } catch(e) {}

  window.store = store; // Сделать доступным глобально для isPremium()

  // Регистрация продуктов
  store.register([
    { id: 'premium_monthly', type: ProductType.PAID_SUBSCRIPTION, platform: Platform.GOOGLE_PLAY },
    { id: 'premium_yearly',  type: ProductType.PAID_SUBSCRIPTION, platform: Platform.GOOGLE_PLAY }
  ]);

  // Колбэки
  store.when()
    .productUpdated((product) => {
      console.log('[billing] product updated:', product.id);
    })
    .approved((transaction) => {
      // approved = Google подтвердил активную подписку
      const productId = transaction.products?.[0]?.id;
      if (productId) {
        activatePremiumPaid(productId);             // обновить profile + setBillingPremium(true)
        window._trustedSetBillingPremium?.(true);
        if (window.systemState) window.systemState.premium = true;
        enqueueBillingStateUpdate(true);
        console.log('[billing] approved:', productId);
      }
      transaction.finish(); // ОБЯЗАТЕЛЬНО — без этого approved будет срабатывать снова
    })
    .verified((receipt) => {
      // verified срабатывает если подключен серверный validator
      const productId = receipt.transactions?.[0]?.products?.[0]?.id;
      if (productId) {
        activatePremiumPaid(productId);
        window._trustedSetBillingPremium?.(true);
        if (window.systemState) window.systemState.premium = true;
        enqueueBillingStateUpdate(true);
      }
      receipt.finish();
    });

  store.error((err) => {
    console.error('[billing] error:', err);
  });

  store.initialize()
    .then(async () => {
      storeReady = true;
      restorePremiumFromProfile(); // Синхронизировать profile → billingPremium флаг

      try {
        // update() — главное: триггерит .approved() для всех активных подписок
        await store.update();
      } catch(e) {
        console.warn('[billing] update failed:', e);
      }

      // Если update() не нашёл покупок — попробовать restore (смена устройства/чистка)
      const monthly = store.get('premium_monthly');
      const yearly  = store.get('premium_yearly');
      const alreadyOwned = (monthly && store.owned(monthly)) || (yearly && store.owned(yearly));
      if (!alreadyOwned) {
        try {
          await store.restorePurchases();
        } catch(e) {
          console.warn('[billing] restorePurchases failed:', e);
        }
      }

      // Финальная проверка и сброс если подписка мертва
      getPremiumFromBilling();
    })
    .catch(err => {
      console.error('[billing] init failed:', err);
    })
    .finally(() => {
      window._billingInitializing = false;
    });
}

export function getPremiumFromBilling() {
  if (!store || !storeReady) return false;
  try {
    const monthly = store.get('premium_monthly');
    const yearly  = store.get('premium_yearly');
    const isPremiumBilling = store.owned(monthly) || store.owned(yearly);

    if (isPremiumBilling) {
      enqueueBillingStateUpdate(true);
      return true;
    }

    // store.owned()=false — store авторитетен, проверить profile
    const profile = getProfile();
    if (profile?.premium_type === 'paid' && profile?.premiumExpiresAt <= Date.now()) {
      // Штамп протух — деактивировать
      console.warn('[billing] profile premium expired — deactivating');
      deactivateExpiredPremium();
    }
    // НЕ доверять profile-fallback когда store.owned()=false и store уже готов.
    // Это ключевое решение: store = источник истины после инициализации.

    return false;
  } catch(e) {
    console.warn('[billing] getPremiumFromBilling error:', e);
    return false;
  }
}

export async function refreshBilling() {
  if (!store || !storeReady) return;
  await store.update();
}

export async function buyMonthly() {
  if (!store || !storeReady) {
    alert('Платёжная система загружается. Попробуйте через несколько секунд.');
    return;
  }
  const product = store.get('premium_monthly', Platform.GOOGLE_PLAY);
  const offer = product?.getOffer();
  if (!offer) { console.warn('[billing] no offer for premium_monthly'); return; }
  const error = await offer.order();
  if (error) console.error('[billing] order error:', error);
}

export async function buyYearly() {
  if (!store || !storeReady) {
    alert('Платёжная система загружается. Попробуйте через несколько секунд.');
    return;
  }
  const product = store.get('premium_yearly', Platform.GOOGLE_PLAY);
  const offer = product?.getOffer();
  if (!offer) { console.warn('[billing] no offer for premium_yearly'); return; }
  const error = await offer.order();
  if (error) console.error('[billing] order error:', error);
}

// Кнопка "Восстановить покупки" в настройках — обязательно по правилам Google Play
export async function restorePurchases() {
  if (!store || !storeReady) return;
  try {
    await store.restorePurchases();
    // После restore .approved() сработает автоматически если подписка активна
  } catch(e) {
    console.warn('[billing] restorePurchases error:', e);
  }
}
```

## Шаг 4. user-profile.js — premium-часть

```js
// ---- PREMIUM ----

export function isPremium() {
  // 1. Runtime-флаг — установлен billing через activatePremiumPaid()
  if (window.__NEYRA_SECURITY__?.billingPremium === true) return true;

  // 2. Store.owned() если store готов (дополнительная проверка)
  try {
    const s = window.store;
    if (s && window._billingInitialized && !window._billingInitializing) {
      const monthly = s.get?.('premium_monthly');
      const yearly  = s.get?.('premium_yearly');
      if ((monthly && s.owned(monthly)) || (yearly && s.owned(yearly))) return true;
    }
  } catch(e) {}

  // 3. Profile как fallback — только пока store ещё загружается
  const profile = getProfile();
  if (!profile) return false;
  if (profile.premium_type === 'paid' && profile.premiumExpiresAt > Date.now()) return true;
  return false;
}

export function setBillingPremium(value) {
  // Обращаться только через защищённый _set если есть
  if (window.__NEYRA_SECURITY__?._set) {
    window.__NEYRA_SECURITY__._set(value === true);
  } else if (window.__NEYRA_SECURITY__) {
    window.__NEYRA_SECURITY__.billingPremium = value === true;
  }
  if (window.systemState) window.systemState.premium = value === true;
}

export function activatePremiumPaid(productId) {
  let profile = getProfile() || {};
  profile.premium = true;
  profile.premium_type = 'paid';
  profile.premiumProductId = productId || 'unknown';
  // 3 дня — штамп обновляется при каждом запуске через store.update() → .approved()
  // Если юзер офлайн 3+ дней и отменил — потеряет доступ. Приемлемо.
  profile.premiumExpiresAt = Date.now() + 3 * 24 * 60 * 60 * 1000;
  profile.premiumActivatedAt = Date.now();
  saveProfile(profile);
  setBillingPremium(true);
}

export function deactivateExpiredPremium() {
  let profile = getProfile();
  if (!profile) return;
  profile.premium = false;
  profile.premium_type = null;
  profile.premiumExpiresAt = null;
  saveProfile(profile);
  setBillingPremium(false);
}

export function restorePremiumFromProfile() {
  // Вызывается сразу после store.initialize() — синхронизировать profile → флаг
  const profile = getProfile();
  if (!profile) return;
  if (profile.premium_type === 'paid' && profile.premiumExpiresAt > Date.now()) {
    setBillingPremium(true);
  } else if (profile.premiumExpiresAt && profile.premiumExpiresAt <= Date.now()) {
    deactivateExpiredPremium(); // Штамп протух — сбросить сразу
  }
}

// Защита от взлома через DevTools
window._trustedSetBillingPremium = setBillingPremium;
Object.defineProperty(window, '__internalPremium', {
  get: () => false,
  set: () => console.warn('Blocked premium hack attempt')
});
```

## Шаг 5. Защита billingPremium флага (app.js или security.js)

```js
// Вызвать до initBilling()
function initSecurityLayer() {
  let _billingPremium = false;

  Object.defineProperty(window, '__NEYRA_SECURITY__', {
    value: {
      get billingPremium() { return _billingPremium; },
      _set(val) { _billingPremium = val === true; }
    },
    writable: false,
    configurable: false
  });
}

initSecurityLayer();
```

Теперь `window.__NEYRA_SECURITY__.billingPremium = true` из консоли не работает — только через `_set`.

## Ключевые решения и почему

| Решение | Почему |
|---|---|
| `store.update()` до `store.restorePurchases()` | `update()` сам триггерит `.approved()` для активных подписок. `restorePurchases()` — только fallback для смены устройства |
| `transaction.finish()` в `.approved()` | Без этого `.approved()` будет вызываться снова при каждом update |
| 3-дневный штамп в profile | Офлайн-работа. Штамп обновляется при каждом запуске если подписка жива. Истёк = деактивация |
| `getPremiumFromBilling()` после update | Если store.owned()=false после update — деактивировать profile не дожидаясь истечения штампа |
| Profile-fallback только пока store загружается | После инициализации store = источник истины. Иначе протухший profile даёт вечный премиум |
| `window.store = store` | Нужен `isPremium()` в user-profile.js для step 2 проверки |

## Известные ловушки

**store.owned() возвращает true при expirationDate=undefined**
Баг плагина (issue #1531). Происходит без серверного validator'а. Митигируется тем что `getPremiumFromBilling()` вызывается после `store.update()` — к тому моменту если подписка мертва `.approved()` не срабатывает, и `getPremiumFromBilling()` деактивирует profile.

**Profile-fallback даёт вечный премиум**
Если в `isPremium()` или `getPremiumFromBilling()` доверять profile когда store.owned()=false — после истечения подписки пользователь остаётся премиум до 3 дней. Решение: не доверять profile когда store уже инициализирован и говорит false.

**Двойной вызов initBilling()**
Защита: `if (window._billingInitialized) return;` в начале функции.

**Продукты не загружаются**
APK не опубликован хотя бы в Internal Testing. Без этого Google Play не отдаёт продукты.

**approved() не срабатывает после покупки**
Gmail аккаунт не добавлен в License Testing в Play Console.
